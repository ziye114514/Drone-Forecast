cmake_minimum_required(VERSION 3.20)
project(local_helix_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED HELIX_SOURCE)
  message(FATAL_ERROR "HELIX_SOURCE must point to the user's existing HelixTrack checkout")
endif()
find_package(Eigen3 CONFIG REQUIRED)
find_package(OpenCV CONFIG REQUIRED COMPONENTS core imgproc)
add_library(helix_reference SHARED helix_bridge.cpp)
target_include_directories(helix_reference PRIVATE "${HELIX_SOURCE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(helix_reference PRIVATE Eigen3::Eigen ${OpenCV_LIBS})
target_compile_definitions(helix_reference PRIVATE NOMINMAX)
